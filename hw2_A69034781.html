You are analyzing reaction time data from a psychology experiment where 20 participants each completed 15 items across two conditions (control and treatment).

  1. (4 points) Create a simulated dataset with the structure described above. Use set.seed(123) for reproducibility.
library(lme4)
set.seed(123)
n_participants <- 20
n_items <- 15
n_conditions <- 2

# Draw participant- and item-level random intercepts from the population
participant_intercepts <- rnorm(n_participants, mean = 0, sd = 80)
item_intercepts        <- rnorm(n_items,        mean = 0, sd = 40)

# Fully crossed design: every participant sees every item in every condition
dat <- expand.grid(
  participant = factor(paste0("P", 1:n_participants)),
  item        = factor(paste0("I", 1:n_items)),
  condition   = factor(c("control", "treatment"))
)

# Assign random effects and generate RT
dat$participant_re <- participant_intercepts[as.numeric(dat$participant)]
dat$item_re        <- item_intercepts[as.numeric(dat$item)]

dat$RT <- 800 +                                          # grand mean
  ifelse(dat$condition == "treatment", -50, 0) +         # fixed treatment effect
  dat$participant_re +                                   # participant variation
  dat$item_re +                                          # item variation
  rnorm(nrow(dat), 0, 50)                                # residual noise

5.2 Parallel Bootstrap (6 points)

  1. (2 points) Create bootstrap_ci_sequential() with parameters: data (numeric vector), n_bootstrap (default 1000), confidence (default 0.95). Return a named vector with lower and upper bounds.
bootstrap_ci_sequential <- function(data, n_bootstrap = 1000, confidence = 0.95) {
  boot_means <- vapply(seq_len(n_bootstrap), function(i) {
    mean(sample(data, length(data), replace = TRUE))
  }, numeric(1))
  
  alpha <- 1 - confidence
  ci <- quantile(boot_means, c(alpha / 2, 1 - alpha / 2))
  c(lower = unname(ci[1]), upper = unname(ci[2]))
}
  1. (2 points) Create bootstrap_ci_parallel() with the same interface but using parallel processing.
bootstrap_ci_parallel <- function(data, n_bootstrap = 1000, confidence = 0.95) {
  cl <- makeCluster(safe_cores)
  # Workers start with empty environments -- must export data
  clusterExport(cl, "data", envir = environment())
  
  boot_means <- unlist(parLapply(cl, seq_len(n_bootstrap), function(i) {
    mean(sample(data, length(data), replace = TRUE))
  }))
  
  stopCluster(cl)
  
  alpha <- 1 - confidence
  ci <- quantile(boot_means, c(alpha / 2, 1 - alpha / 2))
  c(lower = unname(ci[1]), upper = unname(ci[2]))
}
